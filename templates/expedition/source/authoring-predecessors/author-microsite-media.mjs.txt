import fs from 'node:fs';
import {L,C,A,eq,N,I,count,letv,p,k,w,each,E,el,set,get,write,attr,cls,on,html,mount,map,append,observations} from './author-microsite-ui.mjs';
import {math,n,mul,sub,div,put,find} from './author-temple-ui.mjs';
const q={},and=(...x)=>['and',...x],or=(...x)=>['or',...x];
q['media.screens.init']=[p('browser.select',{selector:'[data-screen]'},'screens'),each('$screens','screen',[p('browser.relative',{ref:'$screen.ref',relation:'query',selector:'video'},'video'),w('$video',p('browser.relative',{ref:'$screen.ref',relation:'query',selector:'[data-sound]'},'sound'),w('$sound',on('$sound.ref','click','media.sound.toggle',{video:'$video.ref',sound:'$sound.ref'}),on('$video.ref','volumechange','media.sound.sync',{video:'$video.ref',sound:'$sound.ref'}),k('media.sound.sync',{video:'$video.ref',sound:'$sound.ref'})),p('browser.relative',{ref:'$screen.ref',relation:'parent'},'parent'),p('browser.relative',{ref:'$parent.ref',relation:'query',selector:'.chapters'},'chapters'),w('$chapters',set(C('seek.','$video.ref.handle'),null),on('$chapters.ref','click','media.seek.click',{video:'$video.ref'},{delegate:'[data-seek]'}),on('$video.ref','loadedmetadata','media.seek.loaded',{video:'$video.ref'})))]),p('browser.environment',{},'env'),w('$env.intersection',p('browser.intersection',{selector:'[data-screen] video',query:'media.screens.intersection',input:{},threshold:0.2})),p('browser.select',{selector:'#provenance'},'boxes'),w(N(eq(count('$boxes'),0)),k('media.provenance',{}))];
q['media.sound.sync']=[p('media.read',{ref:'$eventInput.video'},'video'),write('$eventInput.sound','textContent',I('$video.muted','Sound off','Sound on')),attr('$eventInput.sound','aria-pressed',I('$video.muted','false','true'))];
q['media.sound.toggle']=[p('media.read',{ref:'$eventInput.video'},'video'),p('media.write',{ref:'$eventInput.video',name:'muted',value:N('$video.muted')}),w(and('$video.muted',eq('$video.volume',0)),p('media.write',{ref:'$eventInput.video',name:'volume',value:0.8})),k('media.sound.sync',{video:'$eventInput.video',sound:'$eventInput.sound'})];
q['media.screens.intersection']=[each('$eventInput.entries','entry',[w(N('$entry.isIntersecting'),p('media.read',{ref:'$entry.ref'},'video'),w(N('$video.paused'),p('media.pause',{ref:'$entry.ref'})))])];
q['media.seek.click']=[p('number.parse',{value:A('$eventInput.matched.attributes','data-seek')},'time'),set(C('seek.','$eventInput.video.handle'),{time:'$time',phase:'metadata'}),p('media.read',{ref:'$eventInput.video'},'video'),p('number.calculate',{operation:'less',values:L('$video.readyState',1)},'unloaded'),w('$unloaded',p('media.write',{ref:'$eventInput.video',name:'preload',value:'auto'}),p('media.load',{ref:'$eventInput.video'})),w(N('$unloaded'),k('media.seek.prepare',{video:'$eventInput.video',time:'$time'})),p('browser.focus',{ref:'$eventInput.video',preventScroll:true})];
q['media.seek.loaded']=[get(C('seek.','$eventInput.video.handle'),'pending'),w('$pending',w(eq('$pending.phase','metadata'),k('media.seek.prepare',{video:'$eventInput.video',time:'$pending.time'})),w(eq('$pending.phase','blob'),k('media.seek.finish',{video:'$eventInput.video',time:'$pending.time'})))];
q['media.seek.contains']=[{fold:'$eventInput.ranges',as:'range',accumulator:'found',initial:false,steps:[p('number.calculate',{operation:'lessEqual',values:L(A('$range',0),'$eventInput.time')},'after'),p('number.calculate',{operation:'lessEqual',values:L('$eventInput.time',A('$range',1))},'before')],yield:or('$found',and('$after','$before')),into:'result'}];
q['media.seek.prepare']=[p('media.read',{ref:'$eventInput.video'},'video'),k('media.seek.contains',{ranges:'$video.seekable',time:'$eventInput.time'},'seekable'),p('text.matches',{value:'$video.currentSrc',pattern:'^blob:'},'blob'),letv('finish',true),w(and(N('$seekable'),'$video.currentSrc',N('$blob')),{try:[p('asset.blobURL',{url:'$video.currentSrc'},'url'),set(C('seek.','$eventInput.video.handle'),{time:'$eventInput.time',phase:'blob'}),p('media.write',{ref:'$eventInput.video',name:'src',value:'$url'}),p('media.write',{ref:'$eventInput.video',name:'muted',value:'$video.muted'}),letv('finish',false)],catch:[]}),w('$finish',k('media.seek.finish',{video:'$eventInput.video',time:'$eventInput.time'}))];
q['media.seek.finish']=[set(C('seek.','$eventInput.video.handle'),null),p('media.write',{ref:'$eventInput.video',name:'currentTime',value:'$eventInput.time'}),p('media.play',{ref:'$eventInput.video'})];
q['media.provenance']=[{try:[p('asset.json',{url:'manifest.json',cache:'no-cache'},'manifest'),k('media.provenance.view',{manifest:'$manifest'},'node'),...mount('#provenance','$node')],catch:[...mount('#provenance',el('p','fine','The manifest could not be read here. It sits beside the film as ',E('a',{href:'manifest.json'},'manifest.json'),'.'))]}];
q['media.provenance.view']=[letv('m','$eventInput.manifest'),{fold:'$m.files',as:'file',accumulator:'rows',initial:L(),steps:[p('text.replace',{value:'$file.path',pattern:'^films/',flags:'',replacement:''},'path'),...math(div('$file.bytes',1e6),'mb'),p('number.format',{value:'$mb',digits:2},'mb'),p('text.slice',{value:'$file.sha256',start:0,end:16},'hash')],yield:append('$rows',E('tr',{},E('th',{scope:'row'},E('a',{href:'$path'},'$path')),el('td',null,C('$file.kind',I('$file.duration',C(' · ','$file.duration',' s'),''),I('$file.codecs',C(' · ','$file.codecs.video','/','$file.codecs.audio'),''))),el('td',null,C('$mb',' MB')),el('td',null,E('code',{title:'$file.sha256'},C('$hash','…'))))),into:'rows'},letv('result',L(el('dl','prov-facts',...[['Source hash',el('code',null,'$m.sourceHash')],['Defined as','$m.sourceHashDefinition'],['Rendered with',el('code',null,'$m.ffmpeg')],['Duration',C('$m.duration',' s in ',count('$m.scenes'),' scenes')]].map(([title,value])=>el('div',null,el('dt',null,title),el('dd',null,value)))),el('table','prov-table',el('thead',null,el('tr',null,...['File','What','Size','SHA-256'].map(t=>E('th',{scope:'col'},t)))),el('tbody',null,'$rows'))))];
q['media.seek.contains'].push(letv('result',N(N('$result'))));
fs.writeFileSync('algebra/microsite-media.process.json',JSON.stringify({protocol:'temple/surface-process-module/1',queries:observations(q)},null,2)+'\n');
