// Optional observations use ordinary indexed algebra access. The strict seed's
// dotted references intentionally refuse missing fields; do not weaken it.
function observations(value){
 if(typeof value==='string'&&/^\$[A-Za-z_]\w*\./.test(value)){const [first,...rest]=value.split('.');return rest.reduce((v,key)=>A(v,key),first);}
 if(Array.isArray(value)){const items=value.map(observations);if(items[0]==='eq'&&items[2]===null)return ['or',items,['eq',items[1],A({},'absent')]];return items;}
 if(value&&typeof value==='object'){const result=Object.fromEntries(Object.entries(value).map(([key,v])=>[key,observations(v)]));if(result.try&&!result.error)result.error='error';return result;}
 return value;
}
